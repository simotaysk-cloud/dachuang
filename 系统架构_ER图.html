    %% 核心实体关系定义
    BATCHES ||--o{ TRACE_RECORDS : "1:N 包含追踪节点"
    BATCHES ||--o| BLOCKCHAIN_TX : "1:1 / 1:N 锚定防篡改"

    %% 批次总表 (MySQL)
    BATCHES {
        bigint id PK "主键自增"
        string batch_no UK "GS1标准批次号 (核心关联)"
        string product_name "产品名称"
        decimal quantity "重量/体积属性"
        string status "流转状态"
        timestamp created_at "建档时间"
    }

    %% 业务过程记录表 (MySQL)
    TRACE_RECORDS {
        bigint id PK "主键自增"
        string batch_no FK "关联 batches.batch_no"
        string stage "阶段 (种植/加工/质检)"
        string operator_id "操作者"
        json metadata "扩展表单(组装成Hash前的数据)"
        timestamp recorded_at "业务发生时间"
    }

    %% 区块链交易存证底账 (MySQL + EVM)
    BLOCKCHAIN_TX {
        bigint id PK "主键自增"
        string batch_no FK "关联 batches.batch_no"
        string data_hash "哈希值 (聚合上表数据)"
        string tx_hash UK "链上唯一TxID (以太坊)"
        string chain_env "上链网络(Sepolia)"
        timestamp anchored_at "区块打包时间"
    }
